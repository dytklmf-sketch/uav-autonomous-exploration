# PX4 飞控参数配置
# 用于 FAST-LIO + FUEL 自主探索无人机
#
# 设置方法：
# 1. 通过 QGroundControl 手动设置
# 2. 或使用 MAVROS 命令行设置（见下方脚本）

# ============================================
# EKF2 外部定位参数
# ============================================
EKF2_EV_CTRL        15      # 启用外部位置和偏航融合 (xy+z+yaw)
EKF2_HGT_REF        3       # 高度参考源：外部视觉
EKF2_EV_DELAY       0       # 外部视觉延迟（ms）
EKF2_EV_POS_X       0.0     # 外部定位传感器 X 偏移
EKF2_EV_POS_Y       0.0     # 外部定位传感器 Y 偏移
EKF2_EV_POS_Z       0.0     # 外部定位传感器 Z 偏移

# ============================================
# 起飞和降落参数
# ============================================
MIS_TAKEOFF_ALT     0.5     # 自动起飞高度（米）
COM_DISARM_LAND     3.0     # 降落后自动上锁延迟（秒）
MPC_LAND_SPEED      0.3     # 降落速度（m/s）

# ============================================
# 遥控器映射
# ============================================
RC_MAP_FLTMODE      6       # CH6 作为飞行模式切换
# CH5 急停：通过飞行模式实现
# CH6 高位：允许板外控制

# ============================================
# 板外控制参数
# ============================================
COM_OBS_AVOID       0       # 禁用避障（FUEL 自己处理）
COM_RC_OVERRIDE     0       # 禁用 RC 覆盖（保持板外控制优先）

# ============================================
# 位置控制参数
# ============================================
MPC_XY_VEL_MAX      1.5     # 最大水平速度（m/s）
MPC_Z_VEL_MAX_UP    1.0     # 最大上升速度（m/s）
MPC_Z_VEL_MAX_DN    0.8     # 最大下降速度（m/s）
MPC_XY_P            1.0     # 水平位置控制 P 增益
MPC_Z_P             1.0     # 垂直位置控制 P 增益

# ============================================
# 姿态控制参数（可选调整）
# ============================================
MC_ROLL_P           6.5     # 横滚 P 增益
MC_PITCH_P          6.5     # 俯仰 P 增益
MC_YAW_P            2.8     # 偏航 P 增益

# ============================================
# 安全参数
# ============================================
COM_ARM_WO_GPS      1       # 允许无 GPS 解锁
NAV_RCL_ACT         0       # RC 失联动作：禁用（或设为 2=降落）
NAV_DLL_ACT         0       # 数据链路失联动作：禁用（或设为 2=降落）
GF_ACTION           0       # 地理围栏动作：无（或设为 2=降落）

# ============================================
# 电池参数（根据实际电池调整）
# ============================================
BAT_N_CELLS         4       # 电池串数（4S 电池）
BAT_V_EMPTY         3.4     # 单节空电压（V）
BAT_V_CHARGED       4.2     # 单节满电压（V）

# ============================================
# 传感器校准参数（需要在 QGC 中校准）
# ============================================
# - 加速度计校准
# - 陀螺仪校准
# - 磁力计校准（如果使用）
# - 水平校准
# - 遥控器校准
